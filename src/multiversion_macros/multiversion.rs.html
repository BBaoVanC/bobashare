1use crate::dispatcher::{DispatchMethod, Dispatcher};
2use crate::target::Target;
3use proc_macro2::TokenStream;
4use quote::ToTokens;
5use syn::{
6    parenthesized, parse::Parser, parse_quote, punctuated::Punctuated, spanned::Spanned, token,
7    Attribute, Error, ItemFn, LitStr, Meta, ReturnType, Type,
8};
9
10pub(crate) fn make_multiversioned_fn(
11    attr: TokenStream,
12    func: ItemFn,
13) -> Result<TokenStream, syn::Error> {
14    if let ReturnType::Type(_, ty) = &func.sig.output {
15        if let Type::ImplTrait(_) = **ty {
16            return Err(Error::new(
17                ty.span(),
18                "cannot multiversion function with `impl Trait` return type",
19            ));
20        }
21    }
22
23    let mut targets: Option<Vec<Target>> = None;
24    let mut inner_attrs: Option<Vec<Attribute>> = None;
25    let mut dispatcher: Option<DispatchMethod> = None;
26
27    let parser = syn::meta::parser(|meta| {
28        if targets.is_some() && (meta.path.is_ident("targets") || meta.path.is_ident("simd")) {
29            return Err(meta.error("can't specify `targets` or `simd` multiple times"));
30        }
31
32        if meta.path.is_ident("targets") {
33            if meta.input.peek(token::Paren) {
34                let content;
35                parenthesized!(content in meta.input);
36                targets = Some(
37                    Punctuated::<Target, token::Comma>::parse_terminated(&content)?
38                        .into_iter()
39                        .collect(),
40                );
41                return Ok(());
42            } else {
43                let value = meta.value()?;
44                let s: LitStr = value.parse()?;
45
46                if s.value().as_str() == "simd" {
47                    let default_targets = [
48                        // "x86_64+avx512f+avx512bw+avx512cd+avx512dq+avx512vl",
49                        "x86_64+avx2+fma",
50                        "x86_64+sse4.2",
51                        // "x86+avx512f+avx512bw+avx512cd+avx512dq+avx512vl",
52                        "x86+avx2+fma",
53                        "x86+sse4.2",
54                        "x86+sse2",
55                        "aarch64+neon",
56                        // "arm+neon",
57                        // "mips+msa",
58                        // "mips64+msa",
59                        // "powerpc+vsx",
60                        // "powerpc+altivec",
61                        // "powerpc64+vsx",
62                        // "powerpc64+altivec",
63                    ];
64                    targets = Some(
65                        default_targets
66                            .iter()
67                            .map(|x| Target::parse(&LitStr::new(x, meta.path.span())).unwrap())
68                            .collect(),
69                    );
70                    return Ok(());
71                }
72
73                return Err(meta.error("expected a list of features or \"simd\""));
74            }
75        }
76
77        if meta.path.is_ident("attrs") {
78            if inner_attrs.is_some() {
79                return Err(meta.error("can't specify `attrs` multiple times"));
80            }
81            inner_attrs = Some(Vec::new());
82            let content;
83            parenthesized!(content in meta.input);
84            inner_attrs = Some(
85                Punctuated::<Meta, token::Comma>::parse_terminated(&content)?
86                    .into_iter()
87                    .map(|meta| parse_quote! { #[#meta] })
88                    .collect(),
89            );
90            return Ok(());
91        }
92
93        if meta.path.is_ident("dispatcher") {
94            if dispatcher.is_some() {
95                return Err(meta.error("can't specify `dispatcher` multiple times"));
96            }
97            let value = meta.value()?;
98            let s: LitStr = value.parse()?;
99            dispatcher = Some(match s.value().as_str() {
100                "default" => DispatchMethod::Default,
101                "static" => DispatchMethod::Static,
102                "direct" => DispatchMethod::Direct,
103                "indirect" => DispatchMethod::Indirect,
104                _ => {
105                    return Err(meta.error("expected `default`, `static`, `direct`, or `indirect`"))
106                }
107            });
108            return Ok(());
109        };
110
111        Err(meta.error("unrecognized option"))
112    });
113
114    let span = attr.span();
115    parser.parse2(attr)?;
116
117    let targets = if let Some(targets) = targets {
118        for target in targets.iter() {
119            if !target.has_features_specified() {
120                // TODO add span to Target
121                return Err(Error::new(span, "target must have features specified"));
122            }
123        }
124        targets
125    } else {
126        return Err(Error::new(span, "expected `targets`"));
127    };
128
129    let inner_attrs = inner_attrs.unwrap_or_default();
130    let dispatcher = dispatcher.unwrap_or(DispatchMethod::Default);
131
132    Ok(Dispatcher {
133        targets,
134        func,
135        inner_attrs,
136        dispatcher,
137    }
138    .to_token_stream())
139}
